/*
  ============================================================
   Project Hail Mary – Rocky Button Box
   Hardware:
     • ESP32
     • 0.96" SSD1306 OLED  → Rocky image   (SDA=21, SCL=22)
     • 0.91" SSD1306 OLED  → Captions      (SDA=25, SCL=26)
     • DFPlayer Mini        → Audio         (TX2=17, RX2=16)
     • Button               → GPIO 15 (INPUT_PULLUP)
  ============================================================
  Libraries needed (install via Library Manager):
    • Adafruit SSD1306
    • Adafruit GFX
    • DFRobotDFPlayerMini
    • Wire  (built-in)
  ============================================================
*/

#define IMG_SCL          22

#define IMG_HEIGHT       64

// 0.91" OLED  – Captions     (128x32)

// ── DFPlayer ───────────────────────────────────────────────

// ── Button ─────────────────────────────────────────────────
#define BUTTON_PIN       15

// ── Two Wire buses ─────────────────────────────────────────

TwoWire capWire(1);   // I2C bus 1  →  caption OLED

Adafruit_SSD1306 capOLED(CAP_WIDTH, CAP_HEIGHT, &capWire, -1);

// ── DFPlayer ───────────────────────────────────────────────
HardwareSerial dfSerial(2);   // UART2
DFRobotDFPlayerMini dfPlayer;

// ── Caption data ────────────────────────────────────────────
// Each entry: { text, duration_ms }
struct CaptionChunk {
  const char* text;
  unsigned long durationMs;
};

// Track 1 chunks  (timed to match 001.mp3)
const CaptionChunk track1[] = {
  { ".....",1000},
   {"rocky watch crew die,", 3000 },
  { "could not fix",     2000 },
  { "grace say",         1000 },
  { "grace will die,",   3000 },
  { "rocky fix",         2000 }
};
const int TRACK1_CHUNKS = 6;

// Single-line captions for tracks 2-6
const char* singleCaptions[] = {
  "",                     // index 0 unused
  "",                     // index 1 = track1 (handled separately)
  "question",             // track 2
  "fist my bump",         // track 3
  "it is time go",        // track 4
  "Amaze Amaze Amaze!",   // track 5
  "it is not enough"      // track 6
};

// ── State ───────────────────────────────────────────────────
int  currentTrack      = 0;
bool playing           = false;

// Caption scrolling state (track 1)
int  chunkIndex        = 0;
unsigned long chunkStart = 0;

// Single caption hold until audio ends
unsigned long audioEndTime = 0;
unsigned long audioDuration[] = { 0, 11000, 2000, 2000, 2000, 3000, 2000 };
// ^ rough durations in ms for each track (index 1-6)

// Button debounce
unsigned long lastPress  = 0;
const int     DEBOUNCE   = 300;

// ── Helpers ─────────────────────────────────────────────────
void showCaption(const char* txt) {
  capOLED.clearDisplay();
  capOLED.setTextColor(SSD1306_WHITE);
  capOLED.setTextWrap(true);

  // Pick font size: size 2 if text short enough, else size 1
  int len = strlen(txt);
  if (len <= 10) {
    capOLED.setTextSize(2);
    // Centre vertically (32px display, char height ~16px)
    capOLED.setCursor(0, 8);
  } else {
    capOLED.setTextSize(1);
    capOLED.setCursor(0, 12);
  }
  capOLED.print(txt);
  capOLED.display();
}

void clearCaption() {
  capOLED.clearDisplay();
  capOLED.display();
}

void playTrack(int track) {
  currentTrack = track;
  playing      = true;
  chunkIndex   = 0;
  chunkStart   = millis();
  audioEndTime = millis() + audioDuration[track];

  dfPlayer.play(track);

  if (track == 1) {
    showCaption(track1[0].text);
    chunkStart = millis();
  } else {
    showCaption(singleCaptions[track]);
  }
}

// ── Setup ────────────────────────────────────────────────────
void setup() {
  Serial.begin(115200);

  // Button
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  // --- Image OLED (bus 0: SDA=21, SCL=22) ---
  imgWire.begin(IMG_SDA, IMG_SCL, 400000UL);
  if (!imgOLED.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR, false, false)) {
    Serial.println(F("Image OLED not found!"));
    while (true);
  }
  imgOLED.clearDisplay();
  imgOLED.drawBitmap(0, 0, epd_bitmap_71Oie5jy6yL, IMG_WIDTH, IMG_HEIGHT, SSD1306_WHITE);
  imgOLED.display();

  // --- Caption OLED (bus 1: SDA=25, SCL=26) ---
  capWire.begin(CAP_SDA, CAP_SCL, 400000UL);
  if (!capOLED.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR, false, false)) {
    Serial.println(F("Caption OLED not found!"));
    while (true);
  }
  capOLED.clearDisplay();
  capOLED.display();

  // --- DFPlayer ---
  dfSerial.begin(9600, SERIAL_8N1, DF_RX_PIN, DF_TX_PIN);
  delay(1000);
  if (!dfPlayer.begin(dfSerial)) {
    Serial.println(F("DFPlayer not found!"));
    while (true);
  }
  dfPlayer.volume(25);   // 0-30

  randomSeed(analogRead(0));
  Serial.println(F("Rocky ready! Press button."));
}

// ── Loop ─────────────────────────────────────────────────────
void loop() {
  unsigned long now = millis();

  // ── Button press ──────────────────────────────────────────
  if (digitalRead(BUTTON_PIN) == LOW && (now - lastPress) > DEBOUNCE) {
    lastPress = now;

    // Stop any current playback
    dfPlayer.stop();
    playing = false;
    clearCaption();
    delay(100);

    // Pick a random track 1-6
    int track = random(1, 7);
    playTrack(track);
  }

  // ── Track-1 chunk scrolling ───────────────────────────────
  if (playing && currentTrack == 1) {
    unsigned long elapsed = now - chunkStart;
    if (chunkIndex < TRACK1_CHUNKS && elapsed >= track1[chunkIndex].durationMs) {
      chunkIndex++;
      chunkStart = now;
      if (chunkIndex < TRACK1_CHUNKS) {
        showCaption(track1[chunkIndex].text);
      } else {
        clearCaption();
        playing = false;
      }
    }
  }

  // ── Single-caption tracks: clear when audio ends ──────────
  if (playing && currentTrack != 1) {
    if (now >= audioEndTime) {
      clearCaption();
      playing = false;
    }
  }
}
